function toneToCssVar(tone: CategoryData["statusTone"]): string {
  if (tone === "strong" || tone === "above") return "var(--status-good, #12b886)";
  if (tone === "watch") return "var(--status-watch, #F5A623)";
  return "var(--status-bad, #E05C5C)";
}

/** Third overview — org health check: what's filled, what's thin, where to go next. */
function OverviewOrgPulse({
  companyName,
  profileMeta,
  journeyStage,
  cohortLabel,
  categories,
  detailAnswers,
  benchmarkValues,
  documentSlots,
  intelligenceItems = [],
  activeInitiatives = [],
  benchmarkContext,
  runway,
  onOpenTrack,
  onCompleteProfile,
  onEditBenchmark,
  onOpenSources,
  onOpenInitiatives,
}: {
  companyName: string;
  profileMeta?: ScorecardProfileMeta;
  journeyStage?: string;
  cohortLabel?: string;
  categories: CategoryData[];
  detailAnswers: DetailAnswers;
  benchmarkValues: OnboardingBenchmarkInput;
  documentSlots?: ScorecardDocumentSlot[];
  intelligenceItems?: ScorecardIntelligenceItem[];
  activeInitiatives?: ScorecardActiveInitiative[];
  benchmarkContext?: ScorecardBenchmarkContext;
  runway?: number | null;
  onOpenTrack?: (id: ScorecardCategory) => void;
  onCompleteProfile?: () => void;
  onEditBenchmark?: () => void;
  onOpenSources?: () => void;
  onOpenInitiatives?: () => void;
}) {
  const profileProgress = computeProfileAnswerProgress(detailAnswers);
  const benchFilled = METRIC_COHORTS.filter(c => parseMetricValue(benchmarkValues[c.key] ?? "")).length;
  const benchTotal = METRIC_COHORTS.length;
  const docFilled = documentSlots?.filter(slot => slot.current).length ?? 0;
  const docTotal = documentSlots?.length ?? 0;
  const userIntel = intelligenceItems.filter(item => !item.id.startsWith("intel-bench-")).length;
  const benchIntel = intelligenceItems.length - userIntel;
  const avgScore = categories.length
    ? Math.round(categories.reduce((sum, cat) => sum + cat.score, 0) / categories.length)
    : 0;

  const identityChips = [
    profileMeta?.sector?.split("·")[0]?.trim(),
    profileMeta?.headquarters?.trim(),
    profileMeta?.employees?.trim(),
    profileMeta?.domain?.trim(),
  ].filter(Boolean) as string[];

  const contextLines = [
    benchmarkContext?.notableCustomers?.trim(),
    benchmarkContext?.notableHires?.trim(),
    benchmarkContext?.biggestChallenges?.trim(),
  ].filter(Boolean) as string[];

  const pillars = [
    {
      id: "profile" as const,
      label: "Profile depth",
      why: "Company context — improves track scores and AI recommendations.",
      value: profileProgress.percent,
      detail: `${profileProgress.answered} of ${profileProgress.total} fields`,
      action: profileProgress.percent >= 100 ? null : "Complete profile →",
      onAction: onCompleteProfile,
    },
    {
      id: "benchmark" as const,
      label: "Benchmark coverage",
      why: "Peer positioning — cohort bars and gap analysis need your metrics.",
      value: benchTotal > 0 ? Math.round((benchFilled / benchTotal) * 100) : 0,
      detail: `${benchFilled} of ${benchTotal} metrics`,
      action: benchFilled >= benchTotal ? null : "Add benchmark metrics →",
      onAction: onEditBenchmark,
    },
    {
      id: "sources" as const,
      label: "Evidence layer",
      why: "Docs and intel — backs scores with sources beyond form answers.",
      value: Math.min(
        100,
        Math.round(((docFilled + userIntel) / Math.max(1, docTotal + 3)) * 100),
      ),
      detail: `${docFilled} docs · ${userIntel} intel`,
      action: docFilled > 0 && userIntel > 0 ? null : "Add sources →",
      onAction: onOpenSources,
    },
  ];

  const openGaps = pillars.filter(p => p.value < 100 && p.action);
  const weakestTrack = [...categories].sort((a, b) => a.score - b.score)[0];

  return (
    <div className="sc-org-pulse" role="region" aria-label={`${companyName} organization pulse`}>
      <header className="sc-org-pulse-hero">
        <div className="sc-org-pulse-hero-copy">
          <span className="sc-org-pulse-eyebrow">Organization pulse</span>
          <h2 className="sc-org-pulse-title">{companyName}</h2>
          <p className="sc-org-pulse-lead">
            Health check for {companyName} — what Fuel knows today, what&apos;s still thin, and the fastest way to strengthen it.
            {journeyStage ? ` ${journeyStage} stage` : ""}
            {cohortLabel ? ` · ${cohortLabel}` : ""}
            {runway != null ? ` · ~${runway.toFixed(0)} mo runway` : ""}
          </p>
          {identityChips.length > 0 ? (
            <ul className="sc-org-pulse-chips" aria-label="Company attributes">
              {identityChips.map(chip => (
                <li key={chip}>{chip}</li>
              ))}
            </ul>
          ) : null}
        </div>
        <div className="sc-org-pulse-hero-stat" aria-label={`Composite track score ${avgScore} out of 100`}>
          <strong>{avgScore.toString().padStart(2, "0")}</strong>
          <span>Composite</span>
        </div>
      </header>

      <div className="sc-org-pulse-pillars" aria-label="Data coverage">
        {pillars.map(pillar => (
          <div key={pillar.id} className="sc-org-pulse-pillar">
            <div className="sc-org-pulse-pillar-head">
              <span>{pillar.label}</span>
              <strong>{pillar.value.toString().padStart(2, "0")}%</strong>
            </div>
            <p className="sc-org-pulse-pillar-why">{pillar.why}</p>
            <div className="sc-org-pulse-pillar-meter" aria-hidden="true">
              <div className="sc-org-pulse-pillar-fill" style={{ width: `${Math.max(pillar.value, 4)}%` }} />
            </div>
            <span className="sc-org-pulse-pillar-detail">{pillar.detail}</span>
            {pillar.action && pillar.onAction ? (
              <button type="button" className="sc-org-pulse-pillar-action" onClick={pillar.onAction}>
                {pillar.action}
              </button>
            ) : null}
          </div>
        ))}
      </div>

      {openGaps.length > 0 ? (
        <section className="sc-org-pulse-gaps" aria-labelledby="sc-org-pulse-gaps-heading">
          <div className="sc-org-pulse-section-head">
            <h3 id="sc-org-pulse-gaps-heading">Close the gaps</h3>
            <span>{openGaps.length} open</span>
          </div>
          <ul className="sc-org-pulse-gap-list">
            {openGaps.map(gap => (
              <li key={gap.id}>
                <div>
                  <strong>{gap.label}</strong>
                  <span>{gap.detail} — {gap.why}</span>
                </div>
                {gap.onAction ? (
                  <button type="button" className="sc-org-pulse-gap-btn" onClick={gap.onAction}>
                    {gap.action}
                  </button>
                ) : null}
              </li>
            ))}
            {weakestTrack && weakestTrack.score < 70 && onOpenTrack ? (
              <li>
                <div>
                  <strong>{weakestTrack.label} track</strong>
                  <span>Lowest score ({weakestTrack.score}/100) — add {weakestTrack.label} context to lift confidence.</span>
                </div>
                <button type="button" className="sc-org-pulse-gap-btn" onClick={() => onOpenTrack(weakestTrack.id)}>
                  Open {weakestTrack.label} →
                </button>
              </li>
            ) : null}
          </ul>
        </section>
      ) : null}

      <section className="sc-org-pulse-tracks" aria-labelledby="sc-org-pulse-tracks-heading">
        <div className="sc-org-pulse-section-head">
          <h3 id="sc-org-pulse-tracks-heading">Track vitals</h3>
          <span>Click a track to add context</span>
        </div>
        <p className="sc-org-pulse-section-note">Live scores from profile + benchmark data. More answers per track raises confidence.</p>
        <div className="sc-org-pulse-track-grid">
          {categories.map(cat => (
            <button
              key={cat.id}
              type="button"
              className="sc-org-pulse-track-card"
              onClick={() => onOpenTrack?.(cat.id)}
              aria-label={`${cat.label} score ${cat.score} of 100, ${cat.confidenceLevel} confidence`}
            >
              <div className="sc-org-pulse-track-top">
                <span className="sc-org-pulse-track-label">{cat.label}</span>
                <span className={`sc-org-pulse-confidence is-${cat.confidenceTone}`}>{cat.confidenceLevel}</span>
              </div>
              <div className="sc-org-pulse-track-score">
                <strong style={{ color: toneToCssVar(cat.statusTone) }}>{cat.score}</strong>
                <em>/ 100</em>
              </div>
              <div className="sc-org-pulse-track-bar" aria-hidden="true">
                <div
                  className="sc-org-pulse-track-bar-fill"
                  style={{ width: `${cat.score}%`, background: toneToCssVar(cat.statusTone) }}
                />
              </div>
              <p className="sc-org-pulse-track-focus">{cat.glanceFocus || cat.glanceSummary}</p>
              <span className="sc-org-pulse-track-meta">
                {cat.detailAnsweredCount}/{cat.detailQuestionCount} context
                · {cat.intelligenceCount} signals
              </span>
            </button>
          ))}
        </div>
      </section>

      <div className="sc-org-pulse-lower">
        {contextLines.length > 0 ? (
          <section className="sc-org-pulse-context" aria-labelledby="sc-org-pulse-context-heading">
            <h3 id="sc-org-pulse-context-heading">Leadership context</h3>
            <ul>
              {contextLines.map(line => (
                <li key={line}>{line}</li>
              ))}
            </ul>
          </section>
        ) : (
          <section className="sc-org-pulse-context sc-org-pulse-context--empty" aria-labelledby="sc-org-pulse-context-heading">
            <h3 id="sc-org-pulse-context-heading">Leadership context</h3>
            <p>Investor-facing narrative from benchmark (wins, hires, challenges). Complete benchmark to populate.</p>
            {onEditBenchmark ? (
              <button type="button" className="sc-org-pulse-inline-action" onClick={onEditBenchmark}>
                Open benchmark →
              </button>
            ) : null}
          </section>
        )}

        <aside className="sc-org-pulse-aside" aria-label="Workspace inventory">
          <p className="sc-org-pulse-aside-intro">Artifacts Fuel ingests beyond form fields.</p>
          <div className="sc-org-pulse-aside-block">
            <span className="sc-org-pulse-aside-label">Documents</span>
            <strong>{docFilled}<em>/{docTotal || "—"}</em></strong>
            {documentSlots && documentSlots.length > 0 ? (
              <ul className="sc-org-pulse-doc-list">
                {documentSlots.slice(0, 4).map(slot => (
                  <li key={slot.typeId} className={slot.current ? "is-filled" : ""}>
                    {slot.typeLabel}
                    {slot.current ? `: ${slot.current.name}` : ""}
                  </li>
                ))}
              </ul>
            ) : null}
          </div>
          <div className="sc-org-pulse-aside-block">
            <span className="sc-org-pulse-aside-label">Intelligence</span>
            <strong>{intelligenceItems.length}</strong>
            <span className="sc-org-pulse-aside-sub">{userIntel} curated · {benchIntel} from benchmarks</span>
          </div>
          <div className="sc-org-pulse-aside-block">
            <span className="sc-org-pulse-aside-label">Initiatives</span>
            <strong>{activeInitiatives.length}</strong>
            {activeInitiatives[0] ? (
              <span className="sc-org-pulse-aside-sub">{activeInitiatives[0].title}</span>
            ) : (
              <span className="sc-org-pulse-aside-sub">None active yet</span>
            )}
            {onOpenInitiatives ? (
              <button type="button" className="sc-org-pulse-inline-action" onClick={onOpenInitiatives}>
                View initiatives →
              </button>
            ) : null}
          </div>
        </aside>
      </div>
    </div>
  );
}
